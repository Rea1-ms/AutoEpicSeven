def test_claim_closes_reward_and_waits_for_obtained_marker(self):
        activity = object.__new__(FreeGacha20)
        activity.config = MagicMock()
        activity.device = FakeDevice()
        activity.ui_goto = MagicMock()

        frame = {"value": 0}

        def appear(button):
            return button == FREE_20_GACHA_OBTAINED and frame["value"] >= 2

        def click(button, interval=0):
            if button == FREE_20_GACHA and frame["value"] == 0:
                frame["value"] = 1
                return True
            return False

        def close(interval=0):
            if frame["value"] == 1:
                frame["value"] = 2
                return True
            return False

        activity.appear = appear
        activity.appear_then_click = click
        activity.handle_touch_to_close = close
        activity.handle_network_error = MagicMock(return_value=False)

        self.assertTrue(activity.run_claim())
        self.assertEqual(frame["value"], 2)
        self.assertEqual(activity.device.screenshot_count, 2)
        activity.config.task_call.assert_called_once_with(
            "Gacha",
            force_call=False,
        )

def test_checked_state_expires_at_server_update(self):
        checked_at = datetime(2026, 9, 3, 10)
        config = SimpleNamespace(
            Scheduler_ServerUpdate="02:00",
            cross_get=lambda key, default=None: (
                checked_at if key == FREE_GACHA_20_CHECKED_AT else default
            ),
        )

        with patch(
            "tasks.activity.scheduling.get_server_last_update",
            return_value=checked_at - timedelta(hours=8),
        ):
            self.assertTrue(is_free_gacha_20_checked_today(config))

        with patch(
            "tasks.activity.scheduling.get_server_last_update",
            return_value=checked_at + timedelta(hours=16),
        ):
            self.assertFalse(is_free_gacha_20_checked_today(config))
